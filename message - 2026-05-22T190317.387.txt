-- ── safe require ───────────────────────────────────────────────────────────────
local function safeRequire(path, timeout)
    local deadline = os.clock() + (timeout or 10)
    while os.clock() < deadline do
        local ok, r = pcall(require, path)
        if ok and r then return r end
        task.wait(0.1)
    end
    return nil
end

local function getweapon()
    local vm = Workspace:FindFirstChild("ViewModels") if not vm then return nil end
    local fp = vm:FindFirstChild("FirstPerson")       if not fp then return nil end
    for _,c in ipairs(fp:GetChildren()) do
        local p={} for s in c.Name:gmatch("[^-]+") do table.insert(p,s:match("^%s*(.-)%s*$")) end
        if #p>=2 then return p[2] end
    end
end
local function isSling(w) return w and w:lower():find("slingshot",1,true)~=nil end
local function isInfinite() local w=getweapon() return w and INFINITE_AMMO[w]==true end
local function wantSlot() return SLOTS[config.target.weaponPick or "primary"] or 1 end
local function pickMelee() return (config.target.weaponPick or "primary")=="melee" end
local function clampVs(v) return math.clamp(tonumber(v) or 1,0.1,2) end
local function randVs(a,b) a,b=clampVs(a),clampVs(b) if b<a then a,b=b,a end return a==b and a or (a+math.random()*(b-a)) end

local function lfItems() local lf=fighter and fighter.LocalFighter return lf and lf.Items end
local function slotItem(idx)
    local items=lfItems() if not items then return end
    if items[idx] then return items[idx] end
    for _,it in pairs(items) do if it and typeof(it)=="table" then local s=tonumber(it:Get("Slot") or it:Get("Index") or "") if s==idx then return it end end end
end
local function whichSlot(item)
    if not item then return end
    local items=lfItems() local oid=item:Get("ObjectID")
    if items and oid then for i=1,3 do local it=items[i] or items[tostring(i)] if it and it:Get("ObjectID")==oid then return i end end end
end
local function getammo()
    if isInfinite() then return nil,nil,false end
    local lf=fighter and fighter.LocalFighter if not lf or not lf.EquippedItem then return 0,0,pickMelee() end
    local cur,max_,isMelee=0,0,false
    pcall(function()
        local item=lf.EquippedItem local nm=tostring(item:Get("Name") or "")
        cur=item:Get("CurrentAmmo") or item:Get("Ammo") or 0
        max_=item:Get("MaxAmmo") or item:Get("MaxBullets") or cur
        isMelee=MELEE_NMS[nm] or nm:lower():find("melee",1,true)~=nil
    end)
    if pickMelee() then isMelee=true end
    return tonumber(cur) or 0, tonumber(max_) or 0, isMelee
end
local function pressSlot(idx)
    pcall(function()
        local vim=game:GetService("VirtualInputManager")
        local kc=({Enum.KeyCode.One,Enum.KeyCode.Two,Enum.KeyCode.Three})[idx]
        if kc then vim:SendKeyEvent(true,kc,false,game) task.wait(0.02) vim:SendKeyEvent(false,kc,false,game) end
    end)
    if keypress then pcall(keypress,({0x31,0x32,0x33})[idx]) task.wait(0.02) if keyrelease then pcall(keyrelease,({0x31,0x32,0x33})[idx]) end end
end

local rageLastSlotForce=0
local function eqSlot()
    if not config.target.rageMasterOn then return end
    local want=wantSlot() local lf=fighter and fighter.LocalFighter if not lf then return end
    local eq=lf.EquippedItem if eq and whichSlot(eq)==want then return end
    local now=tick() if now-rageLastSlotForce<0.06 then return end rageLastSlotForce=now
    local item=slotItem(want) local oid=item and item:Get("ObjectID")
    if oid then
        local rm=ReplicatedStorage.Remotes.Replication.Fighter.UseItem
        for _,en in ipairs({"Equip","Switch","Select","EquipItem","ChangeItem"}) do pcall(function() local ev=enums:ToEnum(en) if ev then rm:FireServer(oid,ev,nil,nil) end end) end
    end
    pcall(function() if lf.EquipItem then lf:EquipItem(want) end if lf.SwitchToSlot then lf:SwitchToSlot(want) end end)
    pressSlot(want)
end

local function itemReload()
    if isInfinite() then return false end
    local lf=fighter and fighter.LocalFighter if not lf or not lf.EquippedItem then return false end
    local result=false
    pcall(function() local item=lf.EquippedItem for _,k in ipairs({"Reloading","IsReloading","IsReload"}) do local v=item:Get(k) if v==true or v==1 or v=="true" then result=true return end end end)
    if result then return true end
    local st=config.state.reloadStartedAt or 0 return st>0 and (tick()-st)<1.75
end
local function tickAmmo()
    if isInfinite() then config.state.outofammo=false config.state.reloading=false config.state.reloadStartedAt=0 return end
    local cur,_,melee=getammo() local wasOut=config.state.outofammo
    config.state.outofammo=melee and false or ((tonumber(cur) or 0)<=0)
    local reloading=itemReload()
    if reloading and not config.state.reloading then config.state.reloadStartedAt=tick() elseif not reloading then config.state.reloadStartedAt=0 end
    config.state.reloading=reloading
    if config.state.outofammo and not wasOut then if isSling(getweapon()) then config.voidspam.phase="hide" end
    elseif not config.state.outofammo and wasOut then
        if config.voidspam.enabled then config.voidspam.phase="shoot" config.voidspam.lastswitch=tick() else config.voidspam.phase=nil end
    end
end

local function isteam(tp) return player:GetAttribute("TeamID")==tp:GetAttribute("TeamID") end
local function valid(char)
    if not char or not char.Parent then return false end
    local hum=char:FindFirstChildOfClass("Humanoid") if not hum or hum.Health<=0 then return false end
    if not char:FindFirstChild("HumanoidRootPart") then return false end
    local tp=Players:GetPlayerFromCharacter(char) if not tp or isteam(tp) then return false end
    return true
end
local function nearest()
    local cur=UserInputService:GetMouseLocation() local best,bd=nil,math.huge
    for _,p in Players:GetPlayers() do
        if p~=player and p.Character and valid(p.Character) then
            local root=p.Character:FindFirstChild("HumanoidRootPart")
            if root then local sp,on=camera:WorldToViewportPoint(root.Position) if on then local d=(Vector2.new(sp.X,sp.Y)-cur).Magnitude if d<bd then best=p.Character bd=d end end end
        end
    end
    return best
end
local function hitpartfromname(char,name)
    if not char then return nil end
    if name=="Closest" then
        local mp=player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not mp then return char:FindFirstChild("Head") end
        local cl,cd=nil,math.huge
        for _,p in pairs(char:GetChildren()) do if p:IsA("BasePart") then local d=(p.Position-mp.Position).Magnitude if d<cd then cd=d cl=p end end end
        return cl or char:FindFirstChild("Head")
    elseif name=="Random" then
        local pts={"Head","HumanoidRootPart","UpperTorso"} return char:FindFirstChild(pts[math.random(#pts)]) or char:FindFirstChild("Head")
    else return char:FindFirstChild(name) or char:FindFirstChild("Head") end
end
local function updatevel()
    if not config.target.character or not config.prediction.enabled then config.prediction.velocity=Vector3.zero config.prediction.lastposition=nil return end
    local root=config.target.character:FindFirstChild("HumanoidRootPart") if not root then return end
    local now=tick() local dt=now-config.prediction.lasttime
    if dt>0 and dt<0.1 then
        local cur=root.Position
        if config.prediction.lastposition then config.prediction.velocity=config.prediction.velocity:Lerp((cur-config.prediction.lastposition)/dt,0.6) end
        config.prediction.lastposition=cur config.prediction.lasttime=now
    end
end
local function predict(targetpart,origin)
    if not config.prediction.enabled or not targetpart then return targetpart and targetpart.Position or Vector3.zero end
    local bp=targetpart.Position local dist=(bp-origin).Magnitude local ping=0
    pcall(function() ping=game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue()/1000 end)
    return bp+(config.prediction.velocity*((dist/3000+ping)*config.prediction.multiplier))
end
local function canuse()
    local char=player.Character if not char then return false end
    local hum=char:FindFirstChildOfClass("Humanoid") if not hum or hum.Health<=0 then return false end
    if isInfinite() then return true end
    local cur,_,melee=getammo() if cur==nil then return true end return cur>0 or melee
end

local function atkCfg()
    if pickMelee() and not config.target.attackCustomEnabled then return {height=0,front=0,side=0,vertical=-(config.target.meleeFeetDrop or 3.5),radius=0} end
    if config.target.attackCustomEnabled then return {height=config.target.customHeight or 2,front=config.target.customFront or 0,side=config.target.customSide or 0,vertical=config.target.customVertical or 0,radius=config.target.customRadius or 0} end
    local mode=config.target.attackPosition or "default" local weapon=getweapon()
    if mode=="under" then return {height=0,front=0,side=0,vertical=-(config.target.underOffset or 4),radius=0} end
    return {height=(weapon and weapon:lower():find("sniper")) and 8 or 2,front=0,side=0,vertical=0,radius=0}
end
local function setOrbH() config.orbit.height=atkCfg().height end
local function setOrbR() config.orbit.radius=atkCfg().radius or 0 end
local function refreshAtk() setOrbH() setOrbR() end
local function orbBase(root)
    if not root then return Vector3.zero end local s=atkCfg()
    return root.Position+(root.CFrame.LookVector*s.front)+(root.CFrame.RightVector*s.side)+Vector3.new(0,s.vertical,0)
end

local function clrVoidSnap() vfrSnap.cf=nil vfrSnap.lv=nil vfrSnap.av=nil end
local function voidHrp() local c=player.Character return c and c:FindFirstChild("HumanoidRootPart") end
local function rndSkip(mi,ma,dmi,dma) local v=math.random(mi,ma) while v>=dmi and v<=dma do v=math.random(mi,ma) end return v end
local function snapVoid(hrp)
    if not hrp then return end
    vfrSnap.cf=hrp.CFrame vfrSnap.lv=hrp.AssemblyLinearVelocity vfrSnap.av=hrp.AssemblyAngularVelocity
    pcall(function()
        local p=Vector3.new(rndSkip(-vfrLim,vfrLim,-vfrDead,vfrDead),rndSkip(-vfrLim,vfrLim,-vfrDead,vfrDead),rndSkip(-vfrLim,vfrLim,-vfrDead,vfrDead))
        local v=Vector3.new(rndSkip(-vfrLim,vfrLim,-vfrDead,vfrDead),rndSkip(-vfrLim,vfrLim,-vfrDead,vfrDead),rndSkip(-vfrLim,vfrLim,-vfrDead,vfrDead))
        hrp.CFrame=CFrame.new(p)*CFrame.Angles(math.pi,math.pi,math.pi) hrp.AssemblyLinearVelocity=v hrp.AssemblyAngularVelocity=v
    end)
end

RunService:BindToRenderStep("rb_vfr_csync",Enum.RenderPriority.First.Value,function()
    if not vfrSnap.cf then return end local hrp=voidHrp() if not hrp then return end
    pcall(function() hrp.CFrame=vfrSnap.cf hrp.AssemblyLinearVelocity=vfrSnap.lv hrp.AssemblyAngularVelocity=vfrSnap.av end)
end)

local function stopsync()
    if not config.state.csyncactive then return end
    if config.orbit.connection then config.orbit.connection:Disconnect() config.orbit.connection=nil end
    config.state.csyncactive=false config.orbit.active=false clrVoidSnap() config.voidspam.phase=nil
    if config.orbit.savedpos and localfighter and localfighter.Entity and localfighter.Entity.RootPart then
        pcall(function() localfighter.Entity.RootPart.CFrame=config.orbit.savedpos end)
    end
    config.orbit.serverpos=nil config.orbit.savedpos=nil oldpos=nil
end
local function tickVoidSpam()
    if not config.voidspam.enabled then return end
    local now=tick() local el=now-config.voidspam.lastswitch
    if config.voidspam.phase=="shoot" then if el>=config.voidspam.currentduration then config.voidspam.phase="hide" config.voidspam.currentduration=randVs(config.voidspam.hide_min,config.voidspam.hide_max) config.voidspam.lastswitch=now end
    elseif config.voidspam.phase=="hide" then if el>=config.voidspam.currentduration then config.voidspam.phase="shoot" config.voidspam.currentduration=randVs(config.voidspam.shoot_min,config.voidspam.shoot_max) config.voidspam.lastswitch=now end
    else config.voidspam.phase="shoot" config.voidspam.currentduration=randVs(config.voidspam.shoot_min,config.voidspam.shoot_max) config.voidspam.lastswitch=now end
end
local function isHiding() return config.voidspam.enabled and config.voidspam.phase=="hide" end

local function startsync()
    if config.state.csyncactive then return end if config.target.immune then return end
    if not config.target.character or not valid(config.target.character) then return end
    
    eqSlot() config.state.csyncactive=true config.orbit.active=true setOrbH() setOrbR()
    if not config.orbit.savedpos and localfighter and localfighter.Entity and localfighter.Entity.RootPart then config.orbit.savedpos=localfighter.Entity.RootPart.CFrame end
    config.orbit.connection=RunService.Heartbeat:Connect(function(dt)
        if not config.state.csyncactive then return end
        if config.target.immune then stopsync() return end
        if not config.target.character or not valid(config.target.character) then stopsync() return end
        if not localfighter or not localfighter.Entity or not localfighter.Entity.RootPart then stopsync() return end
        setOrbR() oldpos=localfighter.Entity.RootPart.CFrame
        local _w=getweapon() local cs=isSling(_w) eqSlot()
        if not cs and config.voidspam.enabled and not config.target.immune then
            tickVoidSpam()
            if config.voidspam.phase=="hide" then
                config.orbit.serverpos=Vector3.new(math.random(-10000,10000),-99999999999,math.random(-10000,10000))
                pcall(function() localfighter.Entity.RootPart.CFrame=CFrame.new(config.orbit.serverpos) end) return
            end
        end
        if not config.target.character then stopsync() return end
        local root=config.target.character:FindFirstChild("HumanoidRootPart") if not root then stopsync() return end
        if not cs and isHiding() then
            config.orbit.serverpos=Vector3.new(math.random(-10000,10000),-99999999999,math.random(-10000,10000))
            pcall(function() localfighter.Entity.RootPart.CFrame=CFrame.new(config.orbit.serverpos) end) return
        end
        if config.state.outofammo and not cs and _w and not isInfinite() then
            local hrp=voidHrp() if hrp then snapVoid(hrp)
            else config.orbit.serverpos=Vector3.new(math.random(-10000,10000),-99999999999,math.random(-10000,10000))
            pcall(function() localfighter.Entity.RootPart.CFrame=CFrame.new(config.orbit.serverpos) end) end return
        end
        local tp=orbBase(root) config.orbit.angle=config.orbit.angle+(config.orbit.speed*dt)
        config.orbit.serverpos=tp+Vector3.new(math.cos(config.orbit.angle)*config.orbit.radius,config.orbit.height,math.sin(config.orbit.angle)*config.orbit.radius)
        if localfighter and localfighter.Entity and localfighter.Entity.RootPart then
            local cur=localfighter.Entity.RootPart.CFrame local newP=CFrame.new(config.orbit.serverpos)
            if (newP.Position-cur.Position).Magnitude<2000 then localfighter.Entity.RootPart.CFrame=newP config.orbit.savedpos=cur end
        end
    end)
end

local function autoshoot()
    if not config.target.enabled or not config.target.character or not config.target.autoshoot then return end
    if config.target.immune then return end
    local weapon=getweapon() local cs=isSling(weapon)
    if not cs and config.voidspam.enabled and isHiding() then return end
    eqSlot() local lf=fighter and fighter.LocalFighter if not lf or not lf.EquippedItem then return end
    if not isInfinite() then if config.state.reloading or config.state.outofammo then return end if not canuse() then return end end
    local tc=config.target.character if not valid(tc) then return end
    local hp=hitpartfromname(tc,config.target.hitpart) if not hp then return end
    local shootPos = (config.state.csyncactive and config.orbit.serverpos) or (function() local r=player.Character and player.Character:FindFirstChild("HumanoidRootPart") return r and r.Position or Vector3.zero end)()
    local targetPos=predict(hp,shootPos)
    local data={[utf8.char(1)]={[utf8.char(0)]=utility:EncodeCFrame(CFrame.new(shootPos,targetPos)),[utf8.char(1)]=utility:EncodeCFrame(CFrame.new(shootPos,targetPos)),[utf8.char(2)]=hp,[utf8.char(3)]=utility:EncodeCFrame(CFrame.new(0.43,0.25,0.42))}}
    local eq=lf.EquippedItem if eq and eq:Get("ObjectID") then
        ragePerf.lastAttackTick=tick()
        local att=math.clamp(math.floor(tonumber(config.target.shootAttempts) or 1),1,3)
        for _=1,att do pcall(function() ReplicatedStorage.Remotes.Replication.Fighter.UseItem:FireServer(eq:Get("ObjectID"),enums:ToEnum("StartShooting"),data,nil) end) end
    end
end

local immuneList={}
RunService.Heartbeat:Connect(function()
    for _,p in pairs(Players:GetPlayers()) do
        if p==player then continue end
        local char=p.Character local root=char and char:FindFirstChild("HumanoidRootPart")
        local immune=root and root:FindFirstChild("Attachment")~=nil
        if immune and not immuneList[p.Name] then immuneList[p.Name]=true if config.target.lastplayer==p then config.target.immune=true clrVoidSnap() stopsync() end
        elseif not immune and immuneList[p.Name] then immuneList[p.Name]=nil if config.target.lastplayer==p then config.target.immune=false if config.target.enabled and config.target.character and valid(config.target.character) then startsync() end end end
    end
end)

local function cleartarget()
    config.target.enabled=false stopsync() config.target.character=nil config.target.lastchar=nil config.target.lastplayer=nil config.target.manualkey=false config.target.immune=false
end
local function settarget(char)
    if not char then return end local tp=Players:GetPlayerFromCharacter(char)
    config.target.enabled=true config.target.character=char config.target.lastchar=char config.target.lastplayer=tp config.target.immune=false
    local root=char:FindFirstChild("HumanoidRootPart")
    if root and root:FindFirstChild("Attachment") then config.target.immune=true else eqSlot() startsync() end
end
local function togglekey()
    if config.target.enabled then cleartarget() else local t=nearest() if t then config.target.manualkey=true settarget(t) end end
end

local slingConn,slingEnabled=nil,false
local function nearplr()
    local r=player.Character and player.Character:FindFirstChild("HumanoidRootPart") if not r then return end
    local found,best=nil,math.huge
    for _,p in ipairs(Players:GetPlayers()) do
        if p==player or not p.Character then continue end
        local h=p.Character:FindFirstChild("HitboxHead") or p.Character:FindFirstChild("Head") if not h then continue end
        local d=(h.Position-r.Position).Magnitude if d<best then found=h best=d end
    end
    return found
end
local function startSlingTP()
    if slingConn then return end
    slingConn=Workspace.DescendantAdded:Connect(function(d)
        if d.Name=="Slingshot" or d.Name=="CoreProjectile" or d.Name=="OuterProjectile" then
            task.spawn(function()
                task.wait(0.06) local part=d:IsA("BasePart") and d or d:FindFirstChildWhichIsA("BasePart") if not part then return end
                part.CanTouch=true
                for i=1,60 do if not part.Parent then break end local t=nearplr() if t and t:IsA("BasePart") and t.Parent then pcall(firetouchinterest,t,part,0) pcall(firetouchinterest,t,part,1) end task.wait() end
            end)
        end
    end)
end
local function stopSlingTP() if slingConn then slingConn:Disconnect() slingConn=nil end end
local function updatesling()
    local so=isSling(getweapon()) local ht=config.target.enabled and config.target.character~=nil
    if so then
        if not slingEnabled then slingEnabled=true startSlingTP() end
        vhState.active=ht if ht and not config.state.csyncactive then startsync() elseif not ht and config.state.csyncactive and not config.target.enabled then stopsync() end
    else
        if slingEnabled then slingEnabled=false stopSlingTP() end
        vhState.active=false
    end
end

RunService.Heartbeat:Connect(function()
    updatesling() tickAmmo()
    if config.target.auto and not config.target.manualkey then
        local cur,_,melee=getammo() local hasammo=cur==nil or cur>0 or melee
        if hasammo and not config.state.reloading then
            if not config.target.enabled or not config.target.character or not valid(config.target.character) then
                local t=nearest() if t then if config.target.enabled then stopsync() end settarget(t) end
            end
        end
    end
    if config.target.enabled and config.target.character then
        if not valid(config.target.character) then cleartarget() return end
        eqSlot() if config.target.autoshoot and not config.target.immune then autoshoot() end
    end
    if config.target.enabled and config.prediction.enabled then updatevel() end
end)

    if Cache[plr] then return end
    local objects = {
        Box = Drawing.new("Square"),
        Fill = Drawing.new("Square"),
        Tracer = Drawing.new("Line"),
        HealthBarOutline = Drawing.new("Line"),
        HealthBar = Drawing.new("Line"),
        NameTag = Drawing.new("Text"),
        DistanceTag = Drawing.new("Text"),
        ChamHighlight = Instance.new("Highlight")
    }
    objects.Box.Filled = false
    objects.Fill.Filled = true
    objects.NameTag.Size = 13
    objects.NameTag.Center = true
    objects.NameTag.Outline = true
    objects.DistanceTag.Size = 13
    objects.DistanceTag.Center = true
    objects.DistanceTag.Outline = true
    objects.HealthBarOutline.Thickness = 3
    objects.HealthBarOutline.Color = Color3.fromRGB(0, 0, 0)
    objects.HealthBar.Thickness = 1
    objects.ChamHighlight.Enabled = false
    objects.ChamHighlight.Parent = CoreGui
    Cache[plr] = objects
end

local function RemoveESP(plr)
    if Cache[plr] then
        pcall(function()
            for _, obj in pairs(Cache[plr]) do
                if obj.Remove then obj:Remove() elseif obj.Destroy then obj:Destroy() end
            end
        end)
        Cache[plr] = nil
    end
end

RunService.RenderStepped:Connect(function()
    local myCharacter = LocalPlayer.Character
    local myHrp = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")

    for plr, obj in pairs(Cache) do
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")
        local hum = char and char:FindFirstChild("Humanoid")
        local visibleState = false
        
        if plr ~= LocalPlayer and char and hrp and head and hum and hum.Health > 0 and myHrp then
            local distance = (myHrp.Position - hrp.Position).Magnitude
            if distance <= ESP_Config.MaxDistance then
                local hrpPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                if onScreen then
                    visibleState = true
                    local scaleMod = (ESP_Config.BoxScale / 1000)
                    local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                    local legPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                    local boxHeight = math.abs(headPos.Y - legPos.Y) * scaleMod
                    local boxWidth = (boxHeight / 1.5)
                    local boxX = hrpPos.X - (boxWidth / 2)
                    local boxY = headPos.Y - ((boxHeight - math.abs(headPos.Y - legPos.Y)) / 2)

                    if ESP_Config.Box then
                        obj.Box.Size = Vector2.new(boxWidth, boxHeight)
                        obj.Box.Position = Vector2.new(boxX, boxY)
                        obj.Box.Thickness = ESP_Config.BoxThickness
                        obj.Box.Color = ESP_Config.BoxColor
                        obj.Box.Visible = true
                    else obj.Box.Visible = false end

                    if ESP_Config.Fill then
                        obj.Fill.Size = Vector2.new(boxWidth, boxHeight)
                        obj.Fill.Position = Vector2.new(boxX, boxY)
                        obj.Fill.Color = ESP_Config.FillColor
                        obj.Fill.Transparency = ESP_Config.FillTransparency
                        obj.Fill.Visible = true
                    else obj.Fill.Visible = false end

                    if ESP_Config.Tracer then
                        obj.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                        obj.Tracer.To = Vector2.new(hrpPos.X, hrpPos.Y)
                        obj.Tracer.Thickness = ESP_Config.TracerThickness
                        obj.Tracer.Color = ESP_Config.TracerColor
                        obj.Tracer.Visible = true
                    else obj.Tracer.Visible = false end

                    if ESP_Config.Healthbar then
                        local healthPercent = hum.Health / hum.MaxHealth
                        local barX = boxX - 5
                        obj.HealthBarOutline.From = Vector2.new(barX, boxY)
                        obj.HealthBarOutline.To = Vector2.new(barX, boxY + boxHeight)
                        obj.HealthBarOutline.Visible = true
                        obj.HealthBar.From = Vector2.new(barX, boxY + boxHeight)
                        obj.HealthBar.To = Vector2.new(barX, boxY + boxHeight - (boxHeight * healthPercent))
                        obj.HealthBar.Color = Color3.fromRGB(255 - (255 * healthPercent), 255 * healthPercent, 0)
                        obj.HealthBar.Visible = true
                    else
                        obj.HealthBarOutline.Visible = false
                        obj.HealthBar.Visible = false
                    end

                    if ESP_Config.Name then
                        obj.NameTag.Position = Vector2.new(hrpPos.X, boxY - 15)
                        obj.NameTag.Text = plr.DisplayName or plr.Name
                        obj.NameTag.Color = ESP_Config.NameColor
                        obj.NameTag.Visible = true
                    else obj.NameTag.Visible = false end

                    if ESP_Config.Distance then
                        obj.DistanceTag.Position = Vector2.new(hrpPos.X, boxY + boxHeight + 2)
                        obj.DistanceTag.Text = string.format("%d studs", distance)
                        obj.DistanceTag.Color = ESP_Config.DistanceColor
                        obj.DistanceTag.Visible = true
                    else obj.DistanceTag.Visible = false end
                end
            end
        end
        
        if not visibleState then
            for k, v in pairs(obj) do if k ~= "ChamHighlight" then v.Visible = false end end
        end

        if ESP_Config.Chams and char and hum and hum.Health > 0 and myHrp and (myHrp.Position - hrp.Position).Magnitude <= ESP_Config.MaxDistance then
            obj.ChamHighlight.Adornee = char
            obj.ChamHighlight.FillColor = ESP_Config.ChamsColor
            obj.ChamHighlight.OutlineColor = ESP_Config.ChamsColor
            local visualTransparency = 1 - (ESP_Config.ChamsBrightness / 15)
            obj.ChamHighlight.FillTransparency = visualTransparency
            obj.ChamHighlight.OutlineTransparency = math.clamp(visualTransparency - 0.2, 0, 1)
            obj.ChamHighlight.Enabled = true
        else
            obj.ChamHighlight.Enabled = false
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if not ESP_Config.SpoofEnabled then return end
    pcall(function()
        if ESP_Config.LevelSpoof ~= 0 then
            if LocalPlayer:FindFirstChild("CustomLeaderstats") and LocalPlayer.CustomLeaderstats:FindFirstChild("Level") then
                LocalPlayer.CustomLeaderstats.Level.Value = ESP_Config.LevelSpoof
            end
            LocalPlayer:SetAttribute("Level", ESP_Config.LevelSpoof)
        end
        LocalPlayer.Name = ESP_Config.ClientName
        LocalPlayer.DisplayName = ESP_Config.ClientName
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                plr.Name = ESP_Config.EnemyName
                plr.DisplayName = ESP_Config.EnemyName
            end
        end
    end)
end)
